
  codegen:
    # M-020: the typed surface of all five Tier-1 SDKs is generated from the contract; a stale tree fails here.
    name: SDK surface is generated from the contract
    runs-on: ubuntu-24.04
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with: { python-version: "3.13" }
      - name: Generator check and unit tests
        run: |
          python -m pip install pyyaml==6.0.2
          python sangam/sdks/codegen/generate_sdk_surface.py --check
          python -m unittest discover -s sangam/sdks/codegen -p "test_*.py" -v

  tier2-go:
    # M-022 (ADR-D-4 Tier 2): generated Go client (openapi-generator, pinned in generate_tier2.py) + the worker helper.
    name: Tier-2 Go
    runs-on: ubuntu-24.04
    timeout-minutes: 20
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-go@v5
        with: { go-version: "1.22.x", cache: false }
      - name: Worker helper (vet + tests)
        working-directory: sangam/sdks/go
        run: |
          go vet ./...
          go test ./...
      - name: Generate and build the Go client
        run: |
          python3 sangam/sdks/tier2/generate_tier2.py --docker --lang=go
          cd sangam/sdks/tier2/out/go && go build ./...

  tier2-rust:
    name: Tier-2 Rust
    runs-on: ubuntu-24.04
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - name: Worker helper (tests)
        working-directory: sangam/sdks/rust/sangam-worker
        run: cargo test --locked || cargo test
      - name: Generate and build the Rust client
        run: |
          python3 sangam/sdks/tier2/generate_tier2.py --docker --lang=rust
          cd sangam/sdks/tier2/out/rust && cargo build

  tier2-kotlin:
    name: Tier-2 Kotlin
    runs-on: ubuntu-24.04
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with: { distribution: temurin, java-version: "21", cache: maven, cache-dependency-path: "sangam/**/pom.xml" }
      - name: Kotlin extensions over the Java SDK
        working-directory: sangam
        run: |
          mvn -B -ntp -pl sdks/java -am -DskipTests install
          mvn -B -ntp -f sdks/kotlin/pom.xml verify
      - uses: gradle/actions/setup-gradle@v4
        with: { gradle-version: "8.10.2" }
      - name: Generate and build the Kotlin client
        run: |
          python3 sangam/sdks/tier2/generate_tier2.py --docker --lang=kotlin
          gradle -p sangam/sdks/tier2/out/kotlin assemble

  ac08:
    # IX.8 AC-08 (M-023): Java, .NET, Python, Node and PHP workers against one server (H2 profile, compose), same
    # lease, idempotency and error semantics, identical T18 values.
    name: AC-08 cross-language workers
    runs-on: ubuntu-24.04
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with: { distribution: temurin, java-version: "25", cache: maven, cache-dependency-path: "sangam/**/pom.xml" }
      - uses: actions/setup-dotnet@v4
        id: dotnet
        with: { dotnet-version: "8.0.x" }
      - uses: actions/setup-node@v4
        with: { node-version: "22" }
      - uses: actions/setup-python@v5
        with: { python-version: "3.13" }
      - uses: shivammathur/setup-php@v2
        with: { php-version: "8.3", tools: composer:v2, coverage: none }
      - name: Build the server jar and the Java SDK worker
        working-directory: sangam
        run: |
          mvn -B -ntp -pl sangam-server,sdks/java -am -DskipTests package
          mvn -B -ntp -pl sdks/java -DskipTests test-compile dependency:build-classpath -Dmdep.outputFile=target/ac08-cp.txt
      - name: Start the AC-08 server (compose, H2, dev token)
        working-directory: sangam
        run: docker compose -f sdks/conformance/ac08/compose.yaml up -d --build
      - name: Build the other workers
        working-directory: sangam/sdks
        run: |
          (cd dotnet && dotnet new globaljson --force --sdk-version "${{ steps.dotnet.outputs.dotnet-version }}" --roll-forward latestPatch && dotnet build examples/Ac08Worker -c Release)
          python -m pip install -e python
          (cd typescript && npm ci --no-audit --no-fund && npx tsc -p examples/tsconfig.json)
          (cd php && composer install --no-interaction --prefer-dist --no-dev)
      - name: Run the harness
        working-directory: sangam
        env:
          AC08_CMD_JAVA: java -cp java/target/classes:java/target/test-classes:${{ github.workspace }}/sangam/sdks/java/target/ac08-cp.txt.cp org.sangam.sdk.conformance.Ac08Worker
          AC08_CMD_DOTNET: dotnet run --no-build -c Release --project dotnet/examples/Ac08Worker
          AC08_CMD_PYTHON: python python/examples/ac08_worker.py
          AC08_CMD_TYPESCRIPT: node typescript/build-examples/examples/ac08-worker.js
          AC08_CMD_PHP: php php/examples/ac08-worker.php
        run: |
          # the classpath file holds the dependency jars joined by ':'; expand it into the Java command
          cp="$(cat sdks/java/target/ac08-cp.txt)"
          export AC08_CMD_JAVA="java -cp java/target/classes:java/target/test-classes:${cp} org.sangam.sdk.conformance.Ac08Worker"
          python sdks/conformance/ac08/run_ac08.py --base-url http://127.0.0.1:18080 --token ac08-dev-token --require-all
      - name: Server log on failure
        if: failure()
        working-directory: sangam
        run: docker compose -f sdks/conformance/ac08/compose.yaml logs --tail=400
      - name: Stop the server
        if: always()
        working-directory: sangam
        run: docker compose -f sdks/conformance/ac08/compose.yaml down -v
